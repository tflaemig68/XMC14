99  uint32_t timer_id;
100  /* Zeitstempel der letzten Ausfuehrung, keine herunterzaehlenden Timer. */
101  uint32_t last_measurement = 0U;
102  uint32_t last_display = 0U;
103  uint32_t last_led = 0U;
104  uint32_t last_pwm = 0U;
105  /* Platz fuer bis zu 21 Zeichen und das abschliessende Nullzeichen. */
106  char line[22];

109  XMC_GPIO_Init(LED2, &led_config);
110  if (status != DAVE_STATUS_SUCCESS) {
111  XMC_GPIO_SetOutputHigh(LED2);
112  /* Ohne funktionierende Hardware oder Zeitbasis hier bewusst anhalten. */
113  while (1U) { }
114  }

116  * Die zurueckgegebene ID identifiziert den angelegten Softwaretimer. */
117  timer_id = SYSTIMER_CreateTimer(1000U, SYSTIMER_MODE_PERIODIC,
118  milliSec_Handler, NULL);
119  if ((timer_id == 0U) || (SYSTIMER_StartTimer(timer_id) != SYSTIMER_STATUS_SUCCESS)) {

122  while (1U) { }

132  * Originalstruktur beschreiben und das vorhandene DAVE-Handle verwenden. */
133  initTOFSensorData(&TOF_Sensor_1, &I2C_MASTER_0,

193  /* PWM veraendert das Verhaeltnis von Ein- zu Ausschaltzeit am Ausgang.

196  if ((uint32_t)(now - last_pwm) >= 10U) {

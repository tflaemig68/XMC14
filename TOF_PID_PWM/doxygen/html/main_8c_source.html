127 static void TOF_UARTApply(const UARTPROTO_Command_t *requested)
128 {
129  float kp = requested->kp / 100.0f;
130  float ki = requested->ki / 100.0f;
131  float kd = requested->kd / 100.0f;
132  bool source_changed = (UART_Setpoint_mm == 0) != (requested->setpoint == 0);
133  bool group_changed = UART_Group != requested->group;
134  if (source_changed || group_changed) { can_setpoint_valid = false; }
135  if (source_changed || group_changed || height_pid.KP != kp ||
136  height_pid.KI != ki || height_pid.KD != kd) {
137  PIDclear(&height_pid);
138  } else if (UART_Setpoint_mm != requested->setpoint) {
139  /* Ein neues Fuehrungssignal darf den aufgebauten I-Anteil nicht loeschen. */
140  PIDresetDerivative(&height_pid);
141  }
142  UART_Group = requested->group;
143  UART_Setpoint_mm = requested->setpoint;
144  if (UART_Setpoint_mm != 0) { TOF_Setpoint_mm = UART_Setpoint_mm; }
145  else if (!can_setpoint_valid) { TOF_Setpoint_mm = 0; }
146  height_pid.KP = kp; height_pid.KI = ki; height_pid.KD = kd;
147  ++uart_setpoint_count;
148 }

157 static bool TOF_CAN_ApplySetpoint(const CANLIB_Message_t *message)
158 {
159  int32_t height;
160  uint16_t raw;
161  if (message == NULL || UART_Setpoint_mm != 0 || UART_Group < 1U || UART_Group > 32U ||
162  message->id != CANLIB_RX_ID || message->id_bits != CANLIB_ID_BITS ||
163  message->length != 3U || message->dlc != 3U || message->data[0] != UART_Group) {
164  return false;
165  }
166  raw = (uint16_t)((uint16_t)message->data[1] | ((uint16_t)message->data[2] << 8U));
167  height = raw <= INT16_MAX ? (int32_t)raw : (int32_t)raw - 65536L;
168  if (height < UART_SETPOINT_MIN_MM || height > UART_SETPOINT_MAX_MM) { return false; }
169  if (TOF_Setpoint_mm != (int16_t)height) { PIDresetDerivative(&height_pid); }
170  TOF_Setpoint_mm = (int16_t)height;
171  can_setpoint_valid = true;
172  ++can_setpoint_count;
173  return true;
174 }

185 static void TOF_UARTReceive(uint32_t now)
186 {
187  static UARTPROTO_Parser_t parser = {{0}, 0U};
188  static uint32_t last_byte = 0U;
189  if ((uint32_t)(now - last_byte) >= 200U) { parser.count = 0U; }
190  for (uint16_t i = 0U; i < 128U; ++i) {
191  int16_t byte = UARTLIB_ReadByte();
192  UARTPROTO_Command_t requested;
193  if (byte == -1) { break; }
194  if (byte == -2) { parser.count = 0U; continue; }
195  last_byte = now;
196  if (UARTPROTO_Push(&parser, (uint8_t)byte, &requested)) {
197  TOF_UARTApply(&requested);
198  }
199  }
200 }

225 static CANLIB_TxResult_t TOF_CAN_SendDistance(int8_t pwm_percent, uint16_t distance_mm)
226 {
227  static uint8_t message_counter = 0U;
228  uint8_t data[8] = {0};
229  CANLIB_TxResult_t result;
230  if (UART_Group < 1U || UART_Group > 32U) { return CANLIB_TX_ERROR; }
231  data[0] = message_counter & 0x0FU;
232  /* Der Cast uebertraegt auch negative int8_t-Werte als ihr 8-Bit-Muster. */
233  data[1] = (uint8_t)pwm_percent;
234  data[2] = (uint8_t)(distance_mm & 0xFFU);
235  data[3] = (uint8_t)(distance_mm >> 8U);
236  /* Wirksamer Sollwert, -1 solange im CAN-Betrieb noch keine Vorgabe vorliegt. */
237  {
238  uint16_t target = (UART_Setpoint_mm == 0 && !can_setpoint_valid)
239  ? UINT16_MAX : (uint16_t)TOF_Setpoint_mm;
240  data[4] = (uint8_t)target;
241  data[5] = (uint8_t)(target >> 8U);
242  }
243  data[6] = 2U; /* Version 2 meldet zusaetzlich die Sollwertquelle. */
244  data[7] = UART_Setpoint_mm == 0 ? 1U : 0U; /* Bit 0: CAN-Sollwertbetrieb. */
245  result = CANLIB_SendWithId(TOF_CAN_TX_BASE_ID + UART_Group, data, sizeof(data));
246  if (result == CANLIB_TX_ACCEPTED) {
247  message_counter = (uint8_t)((message_counter + 1U) & 0x0FU);
248  ++can_tx_accepted_count;
249  } else if (result == CANLIB_TX_BUSY) {
250  ++can_tx_busy_count;
251  } else {
252  ++can_tx_error_count;
253  }
254  return result;
255 }

278 static uint16_t TOF_MovingAverage(uint16_t distance_mm)
279 {
280  static uint16_t samples[TOF_MEAN_WINDOW_SIZE] = {0};
281  static uint32_t sum = 0U;
282  static uint16_t next_index = 0U;
283  static uint16_t sample_count = 0U;
284 
285  /* Reichweitenfehler sind keine Entfernungen und duerfen die Summe nicht aendern. */
286  if (distance_mm == TOF_VL53L0X_OUT_OF_RANGE) {
287  return TOF_VL53L0X_OUT_OF_RANGE;
288  }
289 
290  if (sample_count < TOF_MEAN_WINDOW_SIZE) {
291  ++sample_count;
292  } else {
293  /* Am Schreibplatz steht bei vollem Puffer der aelteste Messwert. */
294  sum -= samples[next_index];
295  }
296  samples[next_index] = distance_mm;
297  sum += distance_mm;
298 
299  /* Nach dem letzten Feld wieder beim ersten beginnen: ein Ringpuffer. */
300  ++next_index;
301  if (next_index >= TOF_MEAN_WINDOW_SIZE) {
302  next_index = 0U;
303  }
304 
305  /* Halben Divisor addieren: Ganzzahldivision rundet so zum naechsten mm.
306  * sample_count ist hier mindestens 1; eine Division durch 0 ist ausgeschlossen. */
307  return (uint16_t)((sum + sample_count / 2U) / sample_count);
308 }

408 int main(void)
409 {
410  /* const: Diese gemeinsam fuer beide LEDs verwendeten Einstellungen bleiben gleich. */
411  const XMC_GPIO_CONFIG_t led_config = {
412  .mode = XMC_GPIO_MODE_OUTPUT_PUSH_PULL,
413  .output_level = XMC_GPIO_OUTPUT_LEVEL_LOW
414  };
415  /* DAVE_Init() richtet die im Projekt konfigurierten APPs ein. */
416  DAVE_STATUS_t status = DAVE_Init();
417  uint32_t timer_id;
418  /* Zeitstempel der letzten Ausfuehrung, keine herunterzaehlenden Timer. */
419  uint32_t last_measurement = 0U;
420  uint32_t last_display = 0U;
421  uint32_t last_led = 0U;
422  /* Zeitstempel des letzten PWM-Sollwertupdates; Mindestabstand 50 ms. */
423  uint32_t last_pwm = 0U;
424  uint32_t last_valid_measurement = 0U;
425  bool new_measurement = false;
426  int16_t previous_setpoint = TOF_Setpoint_mm;
427  /* CAN bekommt einen eigenen Zeitstempel, unabhaengig von der Anzeige. */
428  uint32_t last_can = 0U;
429  uint32_t last_uart = 0U;
430  bool uart_ready;
431  bool can_ready;
432  CANLIB_TxResult_t can_result = CANLIB_TX_ERROR;
433  /* Gemeinsamer Mittelwert fuer Distanzanzeige und PWM-Sollwert.
434  * distanceFromTOF bleibt der ungefilterte Sensorwert. */
435  uint16_t distance_mean_mm = TOF_VL53L0X_OUT_OF_RANGE;
436  /* Tastgrad in ganzen Prozent (0 bis 100), zugleich Wert der Prozentanzeige.
437  * Erst beim DAVE-Aufruf wird mit 100 auf den Bereich 0 bis 10000 skaliert. */
438  /* Bis zur ersten gueltigen Regelung bleibt die PWM aus. */
439  uint16_t PWM_Value = 0U;
440  /* Platz fuer bis zu 21 Zeichen und das abschliessende Nullzeichen. */
441  char line[22];
442 
443  XMC_GPIO_Init(LED1, &led_config);
444  XMC_GPIO_Init(LED2, &led_config);
445  if (status != DAVE_STATUS_SUCCESS) {
446  XMC_GPIO_SetOutputHigh(LED2);
447  /* Ohne funktionierende Hardware oder Zeitbasis hier bewusst anhalten. */
448  while (1U) { }
449  }
450  PWM_SetDutyCycle(&PWM_0, 0U);
451  PIDinit(&height_pid, HEIGHT_PID_KP, HEIGHT_PID_KI, HEIGHT_PID_KD,
452  PWMset_Periode_ms / 1000.0f);
453  uart_ready = UARTLIB_Init();
454  if (!uart_ready) { XMC_GPIO_SetOutputHigh(LED2); }
455  can_ready = CANLIB_Init();
456  if (!can_ready) {
457  XMC_GPIO_SetOutputHigh(LED2);
458  }
459  /* SYSTIMER erwartet Mikrosekunden: 1000 us entsprechen 1 ms.
460  * Die zurueckgegebene ID identifiziert den angelegten Softwaretimer. */
461  timer_id = SYSTIMER_CreateTimer(1000U, SYSTIMER_MODE_PERIODIC,
462  milliSec_Handler, NULL);
463  if ((timer_id == 0U) || (SYSTIMER_StartTimer(timer_id) != SYSTIMER_STATUS_SUCCESS)) {
464  XMC_GPIO_SetOutputHigh(LED2);
465  /* Ohne funktionierende Hardware oder Zeitbasis hier bewusst anhalten. */
466  while (1U) { }
467  }
468  /* Erst warten, wenn der Zeitzaehler laeuft; Sensor und Display starten mit. */
469  delay_ms(500U);
470  OLED_Init();
471  OLED_Clear(0);
472  TOF_ShowTitle();
473  show_status("Initialisierung ");
474 
475  /* & liefert die Adresse einer Variablen. So kann die Bibliothek die
476  * Originalstruktur beschreiben und das vorhandene DAVE-Handle verwenden. */
477  initTOFSensorData(&TOF_Sensor_1, &I2C_MASTER_0,
478  TOF_ADDR_VL53LOX, TOF_PROFILE, 0U);
479  /* Reihenfolge: Sensor pruefen/kalibrieren, Profil setzen, Dauermessung starten.
480  * Jeder folgende Schritt wird nur nach Erfolg des vorherigen ausgefuehrt. */
481  initedTOF = TOF_init(&TOF_Sensor_1);
482  if (initedTOF) {
483  initedTOF = TOF_set_ranging_profile(&TOF_Sensor_1);
484  }
485  if (initedTOF) {
486  TOF_Sensor_1.enableTOFSensor = true;
487  initedTOF = TOF_start_continuous(&TOF_Sensor_1);
488  }
489  if (!initedTOF) {
490  TOF_Sensor_1.enableTOFSensor = false;
491  ++tof_error_count;
492  XMC_GPIO_SetOutputHigh(LED2);
493  show_status("TOF Init-Fehler ");
494  }
495 
496  last_pwm = SysTickCounter;
497 
498  /* 1U ist immer wahr: Ein Mikrocontroller arbeitet normalerweise dauerhaft. */
499  while (1U) {
500  /* Einmal lesen: Alle Aufgaben dieses Durchlaufs nutzen denselben Zeitwert.
501  * Unsigned-Subtraktion funktioniert auch ueber den Zaehlerueberlauf,
502  * solange die betrachteten Zeitabstaende kleiner als ein Umlauf sind. */
503  uint32_t now = SysTickCounter;
504  if (uart_ready) { TOF_UARTReceive(now); }
505  if (can_ready) {
506  CANLIB_Message_t message;
507  if (CANLIB_Receive(&message)) { (void)TOF_CAN_ApplySetpoint(&message); }
508  }
509  TOF_ShowTitle();
510  /* && verlangt beide Bedingungen: Sensor bereit UND genug Zeit vergangen.
511  * Die 5 ms Zugabe sind ein kleiner Abstand zum vorgesehenen Messbudget. */
512  if (initedTOF && (uint32_t)(now - last_measurement) >=
513  TOF_Sensor_1.Ranging_Profile_time + 5U) {
514  last_measurement = now;
515  /* ! kehrt den Wahrheitswert um: Dieser Zweig behandelt einen Fehler. */
516  if (!TOF_read_continuous_distance(&TOF_Sensor_1)) {
517  ++tof_error_count;
518  initedTOF = false;
519  TOF_Sensor_1.enableTOFSensor = false;
520  XMC_GPIO_SetOutputHigh(LED2);
521  TOF_UpdateScope(TOF_VL53L0X_OUT_OF_RANGE);
522  show_status("TOF Lesefehler ");
523  } else {
524  /* Jede neue Messung genau einmal in die Mittelwertbildung geben. */
525  distance_mean_mm = TOF_MovingAverage(TOF_Sensor_1.distanceFromTOF);
526  if (distance_mean_mm != TOF_VL53L0X_OUT_OF_RANGE) {
527  last_valid_measurement = SysTickCounter;
528  new_measurement = true;
529  }
530  TOF_UpdateScope(TOF_Sensor_1.distanceFromTOF);
531  }
532  }
533  /* Nach Buszugriffen den Zeitwert erneut lesen. Keine Regelung mit alten Daten. */
534  now = SysTickCounter;
535  if ((UART_Setpoint_mm == 0 && !can_setpoint_valid) ||
536  !initedTOF || distance_mean_mm == TOF_VL53L0X_OUT_OF_RANGE ||
537  (uint32_t)(now - last_valid_measurement) >= HEIGHT_MEASUREMENT_TIMEOUT_MS ||
538  (uint32_t)(now - last_pwm) > HEIGHT_CONTROL_TIMEOUT_MS) {
539  PIDclear(&height_pid);
540  PWM_Value = 0U;
541  PWM_SetDutyCycle(&PWM_0, 0U);
542  new_measurement = false;
543  last_pwm = now;
544  } else if (new_measurement && (uint32_t)(now - last_pwm) >= PWMset_Periode_ms) {
545  float output;
546  /* Ein Sollwertsprung soll keinen kuenstlichen D-Impuls ausloesen. */
547  if (TOF_Setpoint_mm != previous_setpoint) { PIDresetDerivative(&height_pid); }
548  previous_setpoint = TOF_Setpoint_mm;
549  /* Sekunden aus ms: Multiplikation statt Software-Float-Division. */
550  height_pid.TA = (float)(uint32_t)(now - last_pwm) * 0.001f;
551  last_pwm = now;
552  new_measurement = false;
553  output = PIDrun(&height_pid, (float)TOF_Setpoint_mm - (float)distance_mean_mm);
554  /* Ganze Prozent fuer CAN/UART; DAVE erwartet Hundertstel-Prozent. */
555  PWM_Value = (uint16_t)(output + 0.5f);
556  PWM_SetDutyCycle(&PWM_0, 100U * PWM_Value);
557  }
558  /* Langsamere Anzeige spart Buszeit; die Messung darf haeufiger erfolgen. */
559  if (!scope_active && (uint32_t)(now - last_display) >= DisplayRate) {
560  last_display = now;
561  if (initedTOF) {
562  if (distance_mean_mm == TOF_VL53L0X_OUT_OF_RANGE) {
563  show_status("Ausser Reichw. ");
564  } else {
565  /* snprintf begrenzt die Ausgabe auf die Puffergroesse.
566  * %5u formatiert eine positive Ganzzahl mit mindestens 5 Stellen. */
567  snprintf(line, sizeof(line), "%5u mm ",
568  (unsigned int)distance_mean_mm);
569  show_status(line);
570  }
571  }
572  snprintf(line, sizeof(line), "Soll:%3dmm PWM:%3u%% ",
573  (int)TOF_Setpoint_mm, (unsigned int)PWM_Value);
574  GUI_ShowString(0, 5, line, 8, 1);
575  /* Auftrag angenommen bedeutet noch nicht, dass ein Empfaenger ACK gab. */
576  /* RX active: gueltiger CAN-Sollwert seit Gruppen-/Quellenwechsel.
577  * Auf 21 Zeichen auffuellen, damit beim Statuswechsel keine Reste bleiben. */
578  {
579  const char *can_status = UART_Group == 0U ? "CAN: keine Gruppe" :
580  !can_ready ? "CAN Init-Fehler" :
581  CANLIB_GetState() == CANLIB_STATE_BUS_OFF ? "CAN Bus-off" :
582  can_result == CANLIB_TX_BUSY ? "CAN TX busy" :
583  can_result == CANLIB_TX_ACCEPTED ?
584  (UART_Setpoint_mm == 0 && can_setpoint_valid ?
585  "CAN TX sent RX active" : "CAN TX sent") : "CAN TX Fehler";
586  snprintf(line, sizeof(line), "%-21s", can_status);
587  GUI_ShowString(0, 6, line, 8, 1);
588  }
589  snprintf(line, sizeof(line), "Zeit: %6lu s", (unsigned long)(now / 1000U));
590  GUI_ShowString(0, OLED_HEIGHT / 8 - 1, line, 8, 1);
591  }
592  if ((uint32_t)(now - last_led) >= 1000U) {
593  last_led = now;
594  XMC_GPIO_ToggleOutput(LED1);
595  }
596  /* Nach dem PWM-Update senden: Byte 1 enthaelt den gesetzten Sollwert. */
597  /* CAN sendet auch bei Sensorfehler einen eindeutig ungueltigen Wert.
598  * Alte Entfernungen werden damit nicht als aktuelle Messung ausgegeben. */
599  if ((uint32_t)(now - last_can) >= TOF_CAN_PERIOD_MS) {
600  uint16_t can_distance = (initedTOF &&
601  distance_mean_mm != TOF_VL53L0X_OUT_OF_RANGE)
602  ? distance_mean_mm : TOF_CAN_INVALID_MM;
603  last_can = now;
604  if (can_ready && UART_Group >= 1U && UART_Group <= 32U) {
605  can_result = TOF_CAN_SendDistance((int8_t)PWM_Value, can_distance);
606  }
607  }
608 
609  /* Acht Bytes: 55 AA, Hoehe, Sollwert, PWM (je int16 Little Endian).
610  * Kein printf und kein Zeilenende auf dieser Leitung. */
611  if (uart_ready && (uint32_t)(now - last_uart) >= UARTLIB_TELEMETRY_MS) {
612  uint8_t packet[8];
613  int16_t height = (initedTOF && distance_mean_mm != TOF_VL53L0X_OUT_OF_RANGE &&
614  distance_mean_mm <= INT16_MAX) ? (int16_t)distance_mean_mm : -1;
615  last_uart = now;
616  UARTPROTO_Encode(packet, height, TOF_Setpoint_mm, (int16_t)PWM_Value);
617  if (!UARTLIB_Write8(packet)) { ++uart_tx_skipped; }
618  }
619 
620  }

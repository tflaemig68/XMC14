308 
309  if (i2cSendByteToSlaveReg(TOF_i2c, TOF_address_used, TOF_REG_SYSTEM_HISTOGRAM_NON, 0x00) != I2C_OK) { return false; }

318

333 bool TOF_set_spads_from_nvm(TOFSensor_t* TOFSENS)
334 {
335  TOF_address_used = TOFSENS->TOF_address_used;
336  TOF_i2c = TOFSENS->i2c_tof;
337 
338  uint8_t spad_count;
339  bool spad_type_is_aperture;
340  if (!TOF_get_spad_info_from_nvm(TOFSENS, &spad_count, &spad_type_is_aperture))
341  {
342  return false;
343  }
344 
345  // The SPAD map (RefGoodSpadMap) is read by VL53L0X_get_info_from_device() in
346  // the API, but the same data seems to be more easily readable from
347  // GLOBAL_CONFIG_SPAD_ENABLES_REF_0 through _6, so read it from there
348  uint8_t ref_spad_map[6];
349  if (i2cBurstRegRead(TOF_i2c, TOF_address_used, TOF_REG_GLOBAL_CONFIG_SPAD_ENABLES_REF_0, ref_spad_map, 6) != I2C_OK) { return false; }
350 
351  if (i2cSendByteToSlaveReg(TOF_i2c, TOF_address_used, TOF_REG_INTERNAL_TUNING_2, 0x01) != I2C_OK) { return false; }
352  if (i2cSendByteToSlaveReg(TOF_i2c, TOF_address_used, TOF_REG_DYNAMIC_SPAD_REF_EN_START_OFFSET, 0x00) != I2C_OK) { return false; }
353  if (i2cSendByteToSlaveReg(TOF_i2c, TOF_address_used, TOF_REG_DYNAMIC_SPAD_NUM_REQUESTED_REF_SPAD, 0x2C) != I2C_OK) { return false; }
354  if (i2cSendByteToSlaveReg(TOF_i2c, TOF_address_used, TOF_REG_INTERNAL_TUNING_2, 0x00) != I2C_OK) { return false; }
355  if (i2cSendByteToSlaveReg(TOF_i2c, TOF_address_used, TOF_REG_GLOBAL_CONFIG_REF_EN_START_SELECT, 0xB4) != I2C_OK) { return false; }
356 
357  uint8_t first_spad_to_enable = spad_type_is_aperture ? 12 : 0; // 12 is the first aperture spad
358  uint8_t spads_enabled = 0;
359 
360  for (uint8_t i = 0; i < 48; i++)
361  {
362  if (i < first_spad_to_enable || spads_enabled == spad_count)
363  {
364  // This bit is lower than the first one that should be enabled, or
365  // (reference_spad_count) bits have already been enabled, so zero this bit
366  ref_spad_map[i / 8] &= ~(1 << (i % 8));
367  }
368  else if ((ref_spad_map[i / 8] >> (i % 8)) & 0x1)
369  {
370  spads_enabled++;
371  }
372  }
373 
374  if (i2cSendByteToSlaveReg(TOF_i2c, TOF_address_used, TOF_REG_GLOBAL_CONFIG_SPAD_ENABLES_REF_0, ref_spad_map[0]) != I2C_OK) { return false; }
375  if (i2cSendByteToSlaveReg(TOF_i2c, TOF_address_used, TOF_REG_GLOBAL_CONFIG_SPAD_ENABLES_REF_0 + 1, ref_spad_map[1]) != I2C_OK) { return false; }
376  if (i2cSendByteToSlaveReg(TOF_i2c, TOF_address_used, TOF_REG_GLOBAL_CONFIG_SPAD_ENABLES_REF_0 + 2, ref_spad_map[2]) != I2C_OK) { return false; }
377  if (i2cSendByteToSlaveReg(TOF_i2c, TOF_address_used, TOF_REG_GLOBAL_CONFIG_SPAD_ENABLES_REF_0 + 3, ref_spad_map[3]) != I2C_OK) { return false; }
378  if (i2cSendByteToSlaveReg(TOF_i2c, TOF_address_used, TOF_REG_GLOBAL_CONFIG_SPAD_ENABLES_REF_0 + 4, ref_spad_map[4]) != I2C_OK) { return false; }
379  if (i2cSendByteToSlaveReg(TOF_i2c, TOF_address_used, TOF_REG_GLOBAL_CONFIG_SPAD_ENABLES_REF_0 + 5, ref_spad_map[5]) != I2C_OK) { return false; }
380 
381  return true;
382 }

693 bool TOF_getMeasurement(TOFSensor_t* TOFSENS, uint16_t *range)
694 {
695  uint32_t poll_start = SysTickCounter;
696  I2C_RETURN_CODE_t i2c_return;
697  TOF_address_used = TOFSENS->TOF_address_used;
698  TOF_i2c = TOFSENS->i2c_tof;
699  uint8_t interrupt_status[1];
700  do
701  {
702  if ((uint32_t)(SysTickCounter - poll_start) >= TOF_SENSOR_TIMEOUT_MS) { return false; }
703  i2c_return = i2cBurstRegRead(TOF_i2c, TOF_address_used, TOF_REG_RESULT_INTERRUPT_STATUS, interrupt_status, 1);

706  {
707  return false;
708  }
709 
710  uint8_t readBuffer[2];
711  i2c_return = i2cBurstRegRead(
712  TOF_i2c, TOF_address_used,
713  TOF_REG_RESULT_RANGE_STATUS + 10,
714  readBuffer, 2);
715  if (i2c_return != I2C_OK)
716  {
717  // returns false, if i2c communication was not successful
718  return false;
719  }
720  *range = (readBuffer[0] << 8) + readBuffer[1];
721 
722  /* Den Zahlenwert speichern, nicht die Adresse des lokalen Bytepuffers. */
723  TOFSENS->measuredRange = *range;
724  i2c_return = i2cSendByteToSlaveReg(
725  TOF_i2c, TOF_address_used,
726  TOF_REG_SYSTEM_INTERRUPT_CLEAR, 0x01);
727  if (i2c_return != I2C_OK)
728  {
729  // returns false, if i2c communication was not successful
730  return false;
731  }
732 
733  /* 8190 or 8191 may be returned when obstacle is out of range. */
734  if (*range == 8190 || *range == 8191)
735  {
736  *range = TOF_VL53L0X_OUT_OF_RANGE;
737  }
738 
739  return true;

40 static I2C_RETURN_CODE_t tof_wait(I2C_MASTER_t *bus, uint32_t flags)
41 {
42  uint32_t start = SysTickCounter;
43  for (;;) {
44  uint32_t status = XMC_I2C_CH_GetStatusFlag(bus->channel);
45  if (status & TOF_I2C_ERRORS) { return I2C_ERROR; }
46  if (status & flags) {
47  XMC_I2C_CH_ClearStatusFlag(bus->channel, flags);
48  return I2C_OK;
49  }
50  if ((uint32_t)(SysTickCounter - start) >= TOF_I2C_TIMEOUT_MS) {
51  return I2C_ERROR;
52  }
53  }
54 }

74 static I2C_RETURN_CODE_t tof_transfer(I2C_MASTER_t *bus, uint16_t address,
75  uint8_t reg, uint8_t *data,
76  uint32_t count, bool read)
77 {
78  I2C_RETURN_CODE_t result;
79  uint32_t nack_event;
80  uint32_t start;
81  if (!bus || !data || !count || address > 0x7fU ||
82  bus->runtime->bus_acquired || bus->runtime->tx_busy || bus->runtime->rx_busy) {
83  return I2C_ERROR;
84  }
85  /* Nur den NACK-Interrupt maskieren, damit er das Fehlerbit nicht vor uns
86  * loescht. Den Timer weiterlaufen lassen: Er wird fuer Zeitlimits gebraucht. */
87  nack_event = bus->channel->PCR_IICMode & XMC_I2C_CH_EVENT_NACK;
88  XMC_I2C_CH_DisableEvent(bus->channel, XMC_I2C_CH_EVENT_NACK);
89  start = SysTickCounter;
90  while (!XMC_USIC_CH_TXFIFO_IsEmpty(bus->channel) ||
91  XMC_USIC_CH_GetTransmitBufferStatus(bus->channel) == XMC_USIC_CH_TBUF_STATUS_BUSY) {
92  if ((uint32_t)(SysTickCounter - start) >= TOF_I2C_TIMEOUT_MS) {
93  result = I2C_ERROR;
94  goto recover;
95  }
96  }
97  XMC_USIC_CH_RXFIFO_Flush(bus->channel);
98  XMC_I2C_CH_ClearStatusFlag(bus->channel, 0xffffffffU);
99  bus->runtime->bus_acquired = true;
100  /* << 1 schafft Platz fuer das Richtungsbit: aus 0x29 wird 0x52. */
101  I2C_MASTER_SendStart(bus, (uint32_t)address << 1, XMC_I2C_CH_CMD_WRITE);
102  result = tof_wait(bus, XMC_I2C_CH_STATUS_FLAG_ACK_RECEIVED);
103  if (result != I2C_OK) { goto recover; }
104  I2C_MASTER_TransmitByte(bus, reg);
105  result = tof_wait(bus, XMC_I2C_CH_STATUS_FLAG_ACK_RECEIVED);
106  if (result != I2C_OK) { goto recover; }
107  if (read) {
108  I2C_MASTER_SendRepeatedStart(bus, (uint32_t)address << 1, XMC_I2C_CH_CMD_READ);
109  result = tof_wait(bus, XMC_I2C_CH_STATUS_FLAG_ACK_RECEIVED);
110  if (result != I2C_OK) { goto recover; }
111  }
112  for (uint32_t i = 0U; i < count; ++i) {
113  if (read) {
114  if (i + 1U == count) { I2C_MASTER_ReceiveNACK(bus); }
115  else { I2C_MASTER_ReceiveACK(bus); }
116  result = tof_wait(bus, XMC_I2C_CH_STATUS_FLAG_RECEIVE_INDICATION |
117  XMC_I2C_CH_STATUS_FLAG_ALTERNATIVE_RECEIVE_INDICATION);
118  if (result != I2C_OK) { goto recover; }
119  data[i] = I2C_MASTER_GetReceivedByte(bus);
120  } else {
121  I2C_MASTER_TransmitByte(bus, data[i]);
122  result = tof_wait(bus, XMC_I2C_CH_STATUS_FLAG_ACK_RECEIVED);
123  if (result != I2C_OK) { goto recover; }
124  }
125  }
126  I2C_MASTER_SendStop(bus);
127  result = tof_wait(bus, XMC_I2C_CH_STATUS_FLAG_STOP_CONDITION_RECEIVED);
128  if (result == I2C_OK) { goto done; }
129 recover:
130  /* goto recover buendelt die Aufraeumarbeit fuer alle Fehlerstellen.
131  * Alte Sendebefehle verwerfen und STOP mit begrenzter Wartezeit versuchen. */
132  XMC_USIC_CH_TXFIFO_Flush(bus->channel);
133  XMC_I2C_CH_ClearStatusFlag(bus->channel, 0xffffffffU);
134  I2C_MASTER_SendStop(bus);
135  (void)tof_wait(bus, XMC_I2C_CH_STATUS_FLAG_STOP_CONDITION_RECEIVED);
136  XMC_USIC_CH_RXFIFO_Flush(bus->channel);
137 done:
138  /* Sowohl Erfolg als auch Fehler geben den Software-Buszustand wieder frei. */
139  bus->runtime->bus_acquired = false;
140  XMC_I2C_CH_ClearStatusFlag(bus->channel, 0xffffffffU);
141  if (nack_event) { XMC_I2C_CH_EnableEvent(bus->channel, XMC_I2C_CH_EVENT_NACK); }
142  return result;
143 }

198 static I2C_RETURN_CODE_t tof_write16(I2C_MASTER_t *bus, uint16_t addr,
199  uint8_t reg, uint16_t value)
200 {
201  /* >> 8 liefert das obere Byte; der uint8_t-Cast behaelt nur 8 Bits. */
202  uint8_t bytes[2] = { (uint8_t)(value >> 8), (uint8_t)value };
203  return tof_transfer(bus, addr, reg, bytes, 2U, false);

214 static I2C_RETURN_CODE_t tof_read16(I2C_MASTER_t *bus, uint16_t addr,
215  uint8_t reg, uint16_t *value)
216 {
217  uint8_t bytes[2];

219  /* *value schreibt an die Adresse des Aufrufers; | verbindet die Bits. */
220  if (status == I2C_OK) { *value = ((uint16_t)bytes[0] << 8) | bytes[1]; }
221  return status;
222 }

72 void IRQ10_Handler(void)
73 {
74  uint32_t errors = XMC_UART_CH_GetStatusFlag(UART_CH) & UART_ERRORS;
75  if (errors || (XMC_USIC_CH_RXFIFO_GetEvent(UART_CH) & XMC_USIC_CH_RXFIFO_EVENT_ERROR)) {
76  ++UARTLIB_RxErrors;
77  rx_fault = true;
78  }
79  XMC_UART_CH_ClearStatusFlag(UART_CH, errors);
80  XMC_USIC_CH_RXFIFO_ClearEvent(UART_CH, RX_FLAGS);
81  for (uint8_t n = 0U; n < 16U && !XMC_USIC_CH_RXFIFO_IsEmpty(UART_CH); ++n) {
82  uint8_t byte = (uint8_t)XMC_USIC_CH_RXFIFO_GetData(UART_CH);
83  uint32_t next = (rx_write + 1U) & RX_MASK;
84  if (next == rx_read) {
85  ++UARTLIB_RxErrors;
86  rx_fault = true;
87  } else {
88  rx_bytes[rx_write] = byte;
89  rx_write = next;
90  }
91  }
92  if (!XMC_USIC_CH_RXFIFO_IsEmpty(UART_CH)) {
93  NVIC_SetPendingIRQ((IRQn_Type)10);
94  }
95 }

97 int16_t UARTLIB_ReadByte(void)
98 {
99  int16_t value = -1;
100  if (!ready) { return -1; }
101  /* Kurze Sperre nur fuer diesen IRQ schuetzt Fehlerbehandlung und Indizes.
102  * SysTick und I2C-Interrupt werden dadurch nicht gesperrt. */
103  NVIC_DisableIRQ((IRQn_Type)10);
104  if (rx_fault) {
105  rx_read = rx_write;
106  rx_fault = false;
107  value = -2;
108  } else if (rx_read != rx_write) {
109  value = rx_bytes[rx_read];
110  rx_read = (rx_read + 1U) & RX_MASK;
111  }
112  NVIC_EnableIRQ((IRQn_Type)10);
113  return value;
114 }

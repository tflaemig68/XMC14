89 bool CANLIB_Receive(CANLIB_Message_t *message)
90 {
91  uint32_t status;
92  if (!initialized || message == NULL || CANLIB_GetState() == CANLIB_STATE_BUS_OFF) {
93  return false;
94  }
95  for (uint8_t attempt = 0U; attempt < 3U; ++attempt) {
96  CANLIB_Message_t snapshot = {0};
97  uint32_t arbitration;
98  uint32_t low;
99  uint32_t high;
100  status = XMC_CAN_MO_GetStatus(&rx_object);
101  if (status & XMC_CAN_MO_STATUS_MESSAGE_LOST) {
102  ++overrun_count;
103  XMC_CAN_MO_ResetStatus(&rx_object, XMC_CAN_MO_RESET_STATUS_MESSAGE_LOST);
104  }
105  if (!(status & XMC_CAN_MO_STATUS_NEW_DATA) ||
106  (status & XMC_CAN_MO_STATUS_RX_UPDATING)) {
107  return false;
108  }
109 
110  /* NEWDAT vor dem Lesen quittieren. Wird es waehrend der Kopie erneut
111  * gesetzt, gehoeren die Register moeglicherweise zu verschiedenen Frames.
112  * Dann die Kopie verwerfen und beim naechsten Versuch erneut lesen. */
113  XMC_CAN_MO_ResetStatus(&rx_object,
114  XMC_CAN_MO_RESET_STATUS_NEW_DATA | XMC_CAN_MO_RESET_STATUS_RX_PENDING);
115  arbitration = rx_object.can_mo_ptr->MOAR;
116  snapshot.dlc = (uint8_t)((rx_object.can_mo_ptr->MOFCR & CAN_MO_MOFCR_DLC_Msk)
117  >> CAN_MO_MOFCR_DLC_Pos);
118  low = rx_object.can_mo_ptr->MODATAL;
119  high = rx_object.can_mo_ptr->MODATAH;
120  status = XMC_CAN_MO_GetStatus(&rx_object);
121  if (status & (XMC_CAN_MO_STATUS_RX_UPDATING | XMC_CAN_MO_STATUS_NEW_DATA)) {
122  continue;
123  }
124  snapshot.id_bits = (arbitration & CAN_MO_MOAR_IDE_Msk) ? 29U : 11U;
125  snapshot.id = (snapshot.id_bits == 11U)
126  ? (arbitration & XMC_CAN_MO_MOAR_STDID_Msk) >> XMC_CAN_MO_MOAR_STDID_Pos
127  : (arbitration & CAN_MO_MOAR_ID_Msk);
128  /* Zusaetzliche Softwarepruefung neben dem Hardwarefilter. */
129  if (snapshot.id != CANLIB_RX_ID || snapshot.id_bits != CANLIB_ID_BITS) {
130  return false;
131  }
132  snapshot.length = snapshot.dlc > 8U ? 8U : snapshot.dlc;
133  for (uint8_t i = 0U; i < snapshot.length; ++i) {
134  snapshot.data[i] = (uint8_t)((i < 4U ? low : high) >> (8U * (i % 4U)));
135  }
136  *message = snapshot;
137  return true;
138  }
139  return false;
140 }

162 CANLIB_TxResult_t CANLIB_SendWithId(uint32_t id, const uint8_t *data, uint8_t length)
163 {
164  if (!initialized || data == NULL || length > 8U || id > CANLIB_ID_MASK ||
165  CANLIB_GetState() == CANLIB_STATE_BUS_OFF) {
166  return CANLIB_TX_ERROR;
167  }
168  /* Unbedingt VOR UpdateData pruefen: Ein laufender Auftrag darf nicht
169  * deaktiviert oder mit teilweise neuen Nutzdaten ueberschrieben werden. */
170  if (XMC_CAN_MO_GetStatus(&tx_object) & XMC_CAN_MO_STATUS_TX_REQUEST) {
171  return CANLIB_TX_BUSY;
172  }
173  XMC_CAN_MO_SetIdentifier(&tx_object, id);
174  tx_object.can_data_length = length;
175  tx_object.can_data[0] = 0U;
176  tx_object.can_data[1] = 0U;
177  for (uint8_t i = 0U; i < length; ++i) {
178  /* Vier Bytes bilden jeweils ein 32-Bit-Hardwareregister.
179  * Explizite Verschiebungen vermeiden Alignment- und Cast-Probleme. */
180  tx_object.can_data[i / 4U] |= (uint32_t)data[i] << (8U * (i % 4U));
181  }
182  XMC_CAN_MO_ResetStatus(&tx_object, XMC_CAN_MO_RESET_STATUS_TX_PENDING);
183  if (XMC_CAN_MO_UpdateData(&tx_object) != XMC_CAN_STATUS_SUCCESS) {
184  return CANLIB_TX_ERROR;
185  }
186  return XMC_CAN_MO_Transmit(&tx_object) == XMC_CAN_STATUS_SUCCESS
187  ? CANLIB_TX_ACCEPTED : CANLIB_TX_ERROR;
188 }
